\begin{posterbox}[before skip=0mm]{Motivation}
  \begin{center}
    \includegraphics[width=0.66\linewidth]{figures/masks_frame80.png}
  \end{center}
  {\normalsize\itshape Class-agnostic 2D masks (CropFormer) in one RGB-D frame
   of Replica \textit{room0}.}\\[3mm]
  \begin{itemize}[leftmargin=14mm, itemsep=2mm]
    \item[\textcolor{tumblue}{\textbullet}] Goal: group per-frame
      \textbf{2D instance masks} into consistent \textbf{3D objects}.
    \item[\textcolor{tumblue}{\textbullet}] MaskClustering (CVPR'24)
      solves this \textbf{purely geometrically} --- no semantics.
    \item[\textcolor{tumblue}{\textbullet}] Can a vision-language model
      (\textbf{Qwen3-VL}) judge ``same object?'' instead --- or in addition?
    \item[\textcolor{accent}{\textbullet}] \textbf{We ablate two axes:}
      the 2D mask backbone (CropFormer vs SAM\,1/SAM\,3) and the edge scorer
      (geometry $\rightarrow$ VLM $\rightarrow$ \textcolor{accent}{\textbf{learned fusion}}).
  \end{itemize}
\end{posterbox}

\begin{posterbox}{Method: One Pipeline, Three Edge Scorers}
  \centering
  \begin{tikzpicture}
    \node[flow] (frames) {RGB-D frames\\[-1mm]{\normalsize Replica scenes}};
    \node[flow, right=14mm of frames] (masks) {2D masks\\[-1mm]{\normalsize CropFormer / SAM}};
    \node[flow, right=14mm of masks] (graph) {Mask graph\\[-1mm]{\normalsize 3D back-projection}};
    \node[flow, fill=accent, below=16mm of graph, minimum width=110mm]
      (score) {EDGE SCORING\\[-1mm]{\normalsize same object? $\rightarrow$ edge weight}};
    \node[flow, fill=tumblue, left=14mm of score, minimum width=90mm]
      (cluster) {Graph clustering\\[-1mm]{\normalsize connected components}};
    \node[flow, fill=tumdark, below=16mm of cluster, minimum width=90mm]
      (objects) {3D object instances};
    \draw[flowarrow] (frames) -- (masks);
    \draw[flowarrow] (masks) -- (graph);
    \draw[flowarrow] (graph) -- (score);
    \draw[flowarrow] (score) -- (cluster);
    \draw[flowarrow] (cluster) -- (objects);
  \end{tikzpicture}\\[7mm]
  \begin{minipage}{0.96\linewidth}
    \normalsize Masks become \textbf{graph nodes}; candidate pairs (3D overlap
    $\geq\tau$) get an edge if the \textcolor{accent}{\textbf{edge score}}
    passes a threshold. Connected components form 3D objects.
    \textbf{We keep the pipeline fixed and swap only the edge scorer.}
  \end{minipage}
\end{posterbox}

\begin{posterbox}[before skip=0mm]{Edge Scoring: Three Approaches}
  \pill{tumlight}{1 \textperiodcentered\ GEOMETRIC (baseline)}\\[4mm]
  {\normalsize \textbf{View consensus:} a mask pair is merged if enough frames that
  observe both also contain them inside one mask (\textit{supporters /
  observers}). Purely geometric --- fails on adjacent objects with high 3D
  overlap.}\\[6mm]

  \pill{tumblue}{2 \textperiodcentered\ SEMANTIC (Qwen3-VL)}\\[4mm]
  \begin{center}
    \includegraphics[width=0.78\linewidth]{figures/qwen_pair.png}
  \end{center}
  {\normalsize\itshape What the VLM sees: two views, one region outlined each
  (A red, B blue).}\\[3mm]
  {\normalsize The VLM answers \textbf{``same physical object?''} with a
  confidence score, used directly as the edge weight. \textbf{Zero-shot} it
  is over-conservative (too few edges, fragmented scenes); \textbf{LoRA
  finetuning} on 5 held-out Replica scenes sharpens it, but raw VLM edges
  stay too sparse to group instances alone.}\\[6mm]

  \pill{accent}{3 \textperiodcentered\ LEARNED FUSION (MLP)}\\[4mm]
  \begin{tcolorbox}[colback=accentpale, colframe=accent, boxrule=1.4pt,
                    arc=4mm, left=8mm, right=8mm, top=6mm, bottom=6mm]
    {\normalsize A small MLP ($2{\times}32$, ${\sim}1.3$k params) combines
    \textbf{geometric features} (view consensus, \#observers, 3D IoU) with the
    \textbf{Qwen confidence} into one edge probability. Trained on 6 scenes,
    validated on 1, tested on \textit{room0}.}\\[4mm]
    {\normalsize \textbf{Question:} does semantics add anything on top of
    geometry? \textcolor{accent}{\textbf{On Replica, barely --- geometry
    dominates.}}}
  \end{tcolorbox}
\end{posterbox}

\begin{posterbox}{Summary \& Outlook}
  \begin{itemize}[leftmargin=12mm, itemsep=2mm]
    \item \textbf{CropFormer is the strongest mask backbone (M2):}
      it beats SAM\,1 and SAM\,3 on downstream 3D AP, so all later
      milestones build on CropFormer masks.
    \item \textbf{The raw VLM signal is weak:} zero-shot and finetuned
      Qwen edges give AP ${\approx}0.02$ --- too sparse to group 3D
      instances on their own.
    \item \textbf{Geometry drives the fusion MLP:} it reaches
      baseline-level AP, but adding the Qwen score moves AP by ${<}0.01$
      --- semantics adds little on top of geometry on Replica.
  \end{itemize}
  \vspace{2mm}
  \begin{tcolorbox}[colback=tumlight!30!white, colframe=tumlight,
                    boxrule=1.4pt, arc=4mm,
                    left=8mm, right=8mm, top=5mm, bottom=5mm]
    {\normalsize\textcolor{tumdark}{\textbf{KEY FINDING}\quad The fusion MLP
    matches the geometric baseline but does not surpass it: on Replica the
    VLM contributes almost no signal once view-consensus geometry is in the
    model.}}\\[3mm]
    {\normalsize\textcolor{tumdark}{\textbf{NEXT STEPS}\quad Stronger VLM
    supervision \,\textperiodcentered\, projection-based single-image prompting
    \,\textperiodcentered\, more scenes and larger VLMs.}}
  \end{tcolorbox}
\end{posterbox}
